# player.gd
# ==========
func _physics_process(delta: float) -> void:
	# Buat flip character
	if direction != 0:
		animated_sprite.flip_h = direction < 0
		

	# Animation
	if not is_on_floor():
		if velocity.y < 0:
			animated_sprite.play("Jump")
		else:
			animated_sprite.play("Fall")
	elif direction != 0:
		animated_sprite.play("Walk")
	else:
		animated_sprite.play("Idle")
